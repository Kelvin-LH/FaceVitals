cmake_minimum_required(VERSION 3.21)
project(FaceVitals VERSION 0.6.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
option(FV_ENABLE_ONNX "Build optional ONNX vision backend" ON)
set(ONNXRUNTIME_ROOT "" CACHE PATH "ONNX Runtime C++ SDK directory")
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Widgets Sql Network Test)
add_library(fv_core
  src/domain/Types.cpp
  src/services/Database.cpp src/services/AuthService.cpp
  src/services/MeasurementRepository.cpp src/services/ModelRegistry.cpp
  src/algorithms/SignalAnalyzer.cpp src/algorithms/EmotionModel.cpp src/algorithms/RoiSignal.cpp
  src/reports/ReportWriter.cpp)
target_include_directories(fv_core PUBLIC src)
target_link_libraries(fv_core PUBLIC Qt6::Core Qt6::Gui Qt6::Sql Qt6::Network)
if(MSVC)
  add_compile_options(/utf-8 /W4)
  target_compile_options(fv_core PUBLIC /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
if(FV_ENABLE_ONNX)
  find_path(ORT_INCLUDE onnxruntime_cxx_api.h PATHS "${ONNXRUNTIME_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
  find_library(ORT_LIBRARY onnxruntime PATHS "${ONNXRUNTIME_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  target_sources(fv_core PRIVATE src/algorithms/OnnxVision.cpp)
  target_include_directories(fv_core PRIVATE "${ORT_INCLUDE}")
  target_link_libraries(fv_core PUBLIC "${ORT_LIBRARY}")
  target_compile_definitions(fv_core PUBLIC FV_HAS_ONNX=1)
endif()
add_library(fv_desktop src/ui/MainWindow.cpp src/ui/PlotWidget.cpp src/ui/EmotionRadar.cpp src/capture/CaptureWorker.cpp src/capture/VideoRecorder.cpp src/capture/SessionArchive.cpp src/capture/CameraFormats.cpp)
if(WIN32)
  target_sources(fv_desktop PRIVATE src/capture/WindowsCamera.cpp src/capture/DirectShowCamera.cpp)
  target_link_libraries(fv_desktop PRIVATE mfplat mfreadwrite mfuuid mf ole32 oleaut32 strmiids)
else()
  target_sources(fv_desktop PRIVATE src/capture/LinuxCamera.cpp)
endif()
target_link_libraries(fv_desktop PUBLIC fv_core Qt6::Widgets)
add_executable(FaceVitals WIN32 src/main.cpp assets/brand.qrc)
if(WIN32)
  target_sources(FaceVitals PRIVATE assets/app.rc)
endif()
target_link_libraries(FaceVitals PRIVATE fv_desktop)
enable_testing()
add_executable(fv_tests tests/TestCore.cpp tests/PulseBaseline.cpp src/training/LocalTrainer.cpp)
target_link_libraries(fv_tests PRIVATE fv_core Qt6::Test Qt6::Widgets)
target_compile_definitions(fv_tests PRIVATE FV_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(fv_desktop_tests tests/TestDesktop.cpp assets/brand.qrc)
target_link_libraries(fv_desktop_tests PRIVATE fv_desktop Qt6::Test)
add_executable(fv_camera_check tests/CameraCheck.cpp)
target_link_libraries(fv_camera_check PRIVATE fv_desktop)
add_executable(fv_replay_check tests/ReplayCheck.cpp tests/PulseBaselineV05.cpp)
target_link_libraries(fv_replay_check PRIVATE fv_core)
add_executable(fv_pulse_tests tests/TestPulseRobustness.cpp)
target_link_libraries(fv_pulse_tests PRIVATE fv_core Qt6::Test)
add_executable(fv_emotion_tests tests/TestEmotionRobustness.cpp)
target_link_libraries(fv_emotion_tests PRIVATE fv_core Qt6::Test)
target_compile_definitions(fv_emotion_tests PRIVATE FV_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
if(WIN32 AND FV_ENABLE_ONNX)
  foreach(target FaceVitals fv_tests fv_desktop_tests fv_camera_check fv_replay_check fv_pulse_tests fv_emotion_tests)
    add_custom_command(TARGET ${target} POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${ONNXRUNTIME_ROOT}/lib/onnxruntime.dll" "$<TARGET_FILE_DIR:${target}>/onnxruntime.dll")
  endforeach()
endif()
add_test(NAME core COMMAND fv_tests)
add_test(NAME pulse_robustness COMMAND fv_pulse_tests)
add_test(NAME emotion_robustness COMMAND fv_emotion_tests)
set_tests_properties(pulse_robustness emotion_robustness PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 180)
set_tests_properties(core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_test(NAME desktop_smoke COMMAND FaceVitals --smoke --assets "${CMAKE_CURRENT_SOURCE_DIR}/models/bootstrap")
set_tests_properties(desktop_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 30)
add_test(NAME desktop_lifecycle COMMAND fv_desktop_tests -o desktop-tests.txt,txt)
set_tests_properties(desktop_lifecycle PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 60)
install(TARGETS FaceVitals RUNTIME DESTINATION bin)
